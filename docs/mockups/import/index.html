// Imported from a Studio 5000 L5X by QuickSFC.

program MainProgram

tag Bonsens : BOOL = 1
tag T1 : TIMER 1s500ms
tag T10 : TIMER 1s500ms
tag T13 : TIMER 500ms
tag T17 : TIMER 500ms
tag T2 : TIMER 3s
tag T3 : TIMER 1s500ms
tag T4 : TIMER 500ms
tag T7 : TIMER 1s250ms
tag T9 : TIMER 1s500ms
tag X : DINT = 1
tag Y : DINT
tag Y5a : BOOL
tag Y5b : BOOL

io
  a0       Local:1:I.Data.4
  a1       Local:1:I.Data.5
  A_moins  Local:1:O.Data.2
  A_plus   Local:1:O.Data.1
  B        Local:1:O.Data.3
  b0       Local:1:I.Data.6
  b1       Local:1:I.Data.7
  C        Local:1:O.Data.5
  D        Local:1:O.Data.6
  d1       Local:1:I.Data.9
  E        Local:1:O.Data.7
  e1       Local:1:I.Data.10
  F        Local:1:O.Data.8
  f0       Local:1:I.Data.11
  f1       Local:1:I.Data.12
  L        Local:1:O.Data.9
  pp       Local:1:I.Data.13
  reset    Local:1:I.Data.3
  start    Local:1:I.Data.0
  stop     Local:1:I.Data.1
  V        Local:1:O.Data.4
end

sfc SFC
init s0
  L[:=]1;
-- pp and f0 and NOT(d1) and NOT(e1) and start
step s1
  F:=1;
-- f1
step s2 3s
  D:=1;
select
  -- s2.DN and d1
  step s3 1s500ms
    Bonsens:=1;
  -- s3.DN
|
  -- tr10: s2.DN and not d1
  step s9 1s500ms
    Bonsens:=0;
  -- tr20: s9.DN
end
step s4 500ms
  D:=0;
-- tr4: s4.DN
step s5
  F:=0;
-- tr5: f0
step s6
  E:=1;
-- tr6: e1
step s7 1s250ms
-- tr7: s7.DN
step s8
  E:=0;
select
  -- tr9: Bonsens
  step s11
    A_moins[:=]1;
  -- tr12: a0
  step s12
    B:=1;
  -- tr13: b1
  step s13 500ms
    V:=1;
  -- tr14: s13.DN
  step s14
    B:=0;
  -- tr15: b0
  step s15
    A_plus[:=]1;
  -- tr16: a1
  step s16
    B:=1;
  -- tr17: b1
  step s17 500ms
    V:=0;
  -- tr18: s17.DN
  step s18
    B:=0;
  -- tr19: b0
  step s19
    A_moins[:=]1;
  -- tr21: a0 -> s0
|
  -- tr8: not Bonsens
  step s10 1s500ms
    C[:=]1;
  -- tr11: s10.DN -> s0
end

Opens as a new file, MainProgram.qsfc (nothing is saved yet). Export it again with File → Export L5X.
  • Several programs in the file (a whole-controller export): the Program list at the top picks one.
  • If the editor has unsaved changes, the usual "Discard changes?" question comes first.
  • The CLI does the same: qsfc import MainProgram.L5X -o MainProgram.qsfc [--program NAME].
